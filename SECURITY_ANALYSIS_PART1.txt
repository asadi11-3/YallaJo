public sealed class RegisterCommandHandler(
    IUserRegistrationService userRegistrationService,
    IProfileCreationService profileCreationService)
    : ICommandHandler<RegisterCommand, RegisterResult>
{
    public async Task<Result<RegisterResult>> Handle(
        RegisterCommand request,
        CancellationToken cancellationToken)
    {
        var result = await userRegistrationService.RegisterAsync(
            new UserRegistrationRequest(
                request.FirstName,
                request.LastName,
                request.Email,
                request.Password),
            cancellationToken);

        if (result.IsFailure)
        {
            return Result<RegisterResult>.Fail(
                result.Outcome,
                result.Messages.FirstOrDefault() ?? string.Empty,
                result.Errors.ToArray());
        }

        var userId = result.Value;

        var profileResult = await profileCreationService.CreateForUserAsync(
            new ProfileCreationRequest(
                userId,
                request.FirstName,
                request.LastName),
            cancellationToken);

        if (profileResult.IsFailure && profileResult.Outcome != Outcome.Conflict)
        {
            return Result<RegisterResult>.Fail(
                profileResult.Outcome,
                profileResult.Messages.FirstOrDefault() ?? string.Empty,
                profileResult.Errors.ToArray());
        }

        return Result<RegisterResult>.Created(new RegisterResult(userId));
    }
}

KEY POINTS:
- Delegates to IUserRegistrationService.RegisterAsync() (Security module)
- Creates profile via IProfileCreationService.CreateForUserAsync() (Accounts module)
- Does NOT directly persist User or send OTP
- OTP is sent by Auth module AFTER registration via ResendOtp
